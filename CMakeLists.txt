cmake_minimum_required(VERSION 3.24)
project(lmcad LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMCAD_FULL_STATIC "Link the complete executable statically" ON)
set(LMCAD_OCCT_PREFIX "" CACHE PATH "Prefix containing a static OpenCASCADE install")
if(NOT IS_DIRECTORY "${LMCAD_OCCT_PREFIX}/include/opencascade" OR
   NOT EXISTS "${LMCAD_OCCT_PREFIX}/lib/cmake/opencascade/OpenCASCADEConfig.cmake")
  message(FATAL_ERROR "Set LMCAD_OCCT_PREFIX to an OCCT 8.0.1 static install (include/opencascade and lib/cmake/opencascade are required): '${LMCAD_OCCT_PREFIX}'")
endif()
# Do not silently pick an unrelated cached or package-registry build.
set(OpenCASCADE_DIR "${LMCAD_OCCT_PREFIX}/lib/cmake/opencascade")
find_package(OpenCASCADE 8.0.1 EXACT CONFIG REQUIRED NO_DEFAULT_PATH PATHS "${OpenCASCADE_DIR}")
set(LMCAD_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmcad.dsl.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LMCAD_SCHEMA}")
file(READ "${LMCAD_SCHEMA}" LMCAD_SCHEMA_HEX HEX)
string(LENGTH "${LMCAD_SCHEMA_HEX}" LMCAD_SCHEMA_HEX_LENGTH)
math(EXPR LMCAD_SCHEMA_LAST_BYTE "${LMCAD_SCHEMA_HEX_LENGTH} - 2")
set(LMCAD_SCHEMA_INITIALIZER "")
foreach(offset RANGE 0 ${LMCAD_SCHEMA_LAST_BYTE} 2)
  string(SUBSTRING "${LMCAD_SCHEMA_HEX}" ${offset} 2 byte)
  string(APPEND LMCAD_SCHEMA_INITIALIZER "0x${byte},")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmcad { inline constexpr unsigned char embedded_schema[] = {${LMCAD_SCHEMA_INITIALIZER}}; inline constexpr std::size_t embedded_schema_size = sizeof(embedded_schema); }\n")
add_executable(lmcad cpp/main.cpp cpp/cad.cpp cpp/scene.cpp cpp/render.cpp cpp/image.cpp cpp/export.cpp)
target_include_directories(lmcad PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(lmcad SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_compile_options(lmcad PRIVATE -Wall -Wextra -Wpedantic)
set(LMCAD_OCCT_LIBS TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKShHealing TKPrim TKBO TKBool TKFillet TKOffset TKMesh)
# GNU ld needs --start-group for the circular OCCT archive refs; Apple's ld64
# re-scans archives automatically; MinGW uses GNU ld but has no libdl/librt.
if(APPLE)
  set(LMCAD_LINK_GROUP ${LMCAD_OCCT_LIBS})
  set(LMCAD_SYS_LIBS "")
elseif(WIN32)
  set(LMCAD_LINK_GROUP "-Wl,--start-group" ${LMCAD_OCCT_LIBS} "-Wl,--end-group")
  set(LMCAD_SYS_LIBS "")
else()
  set(LMCAD_LINK_GROUP "-Wl,--start-group" ${LMCAD_OCCT_LIBS} "-Wl,--end-group")
  set(LMCAD_SYS_LIBS pthread dl m rt)
endif()
target_link_libraries(lmcad PRIVATE ${LMCAD_LINK_GROUP} ${LMCAD_SYS_LIBS})
add_executable(lmcad-unit tests/unit.cpp cpp/cad.cpp cpp/scene.cpp cpp/render.cpp cpp/image.cpp cpp/export.cpp)
target_include_directories(lmcad-unit PRIVATE cpp)
target_include_directories(lmcad-unit SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_compile_options(lmcad-unit PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(lmcad-unit PRIVATE ${LMCAD_LINK_GROUP} ${LMCAD_SYS_LIBS})
add_executable(lmcad-apng-extract tools/apng-extract.cpp)
# "Clang" also matches AppleClang; macOS has no -static/-static-libstdc++.
# MinGW (GNU) takes the GNU static flags, which produce a standalone .exe.
if((CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang") AND NOT APPLE)
  target_link_options(lmcad PRIVATE -static-libstdc++ -static-libgcc)
  if(LMCAD_FULL_STATIC)
    target_link_options(lmcad PRIVATE -static)
  endif()
endif()
