cmake_minimum_required(VERSION 3.24)
project(lmcad LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(LMCAD_FULL_STATIC "Link the complete executable statically" OFF)
set(LMCAD_SCHEMA "${CMAKE_CURRENT_SOURCE_DIR}/schema/lmcad.dsl.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${LMCAD_SCHEMA}")
file(READ "${LMCAD_SCHEMA}" LMCAD_SCHEMA_HEX HEX)
string(LENGTH "${LMCAD_SCHEMA_HEX}" LMCAD_SCHEMA_HEX_LENGTH)
math(EXPR LMCAD_SCHEMA_LAST_BYTE "${LMCAD_SCHEMA_HEX_LENGTH} - 2")
set(LMCAD_SCHEMA_INITIALIZER "")
foreach(offset RANGE 0 ${LMCAD_SCHEMA_LAST_BYTE} 2)
  string(SUBSTRING "${LMCAD_SCHEMA_HEX}" ${offset} 2 byte)
  string(APPEND LMCAD_SCHEMA_INITIALIZER "0x${byte},")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/schema_embedded.hpp"
  "#pragma once\n#include <cstddef>\nnamespace lmcad { inline constexpr unsigned char embedded_schema[] = {${LMCAD_SCHEMA_INITIALIZER}}; inline constexpr std::size_t embedded_schema_size = sizeof(embedded_schema); }\n")
add_executable(lmcad cpp/main.cpp cpp/scene.cpp cpp/render.cpp cpp/image.cpp cpp/export.cpp)
target_include_directories(lmcad PRIVATE cpp "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_options(lmcad PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lmcad-unit tests/unit.cpp cpp/scene.cpp cpp/render.cpp cpp/image.cpp cpp/export.cpp)
target_include_directories(lmcad-unit PRIVATE cpp)
target_compile_options(lmcad-unit PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_link_options(lmcad PRIVATE -static-libstdc++ -static-libgcc)
  if(LMCAD_FULL_STATIC)
    target_link_options(lmcad PRIVATE -static)
  endif()
endif()
